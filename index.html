// ============================================================ // CLASIFICACIÓN DE MERCADERÍA (Cambios y rotura) // ============================================================ const SHEET_CLASIFICACION = "Cambios y rotura"; function guardarClasificacion(codigo, producto, categoria, cantidad, observaciones, timestamp) { codigo = String(codigo || "").trim(); producto = String(producto || "").trim(); categoria = String(categoria || "").trim().toLowerCase(); cantidad = Number(cantidad) || 1; observaciones = String(observaciones || "").trim(); timestamp = timestamp || new Date().toISOString(); if (!codigo) return { ok: false, error: "Falta el código" }; if (!categoria) return { ok: false, error: "Falta la categoría" }; if (!["roto", "vencido", "cambio", "comedor"].includes(categoria)) { return { ok: false, error: "Categoría inválida" }; } const sheet = getSheet_(SHEET_CLASIFICACION); if (sheet.getLastRow() === 0) { sheet.appendRow([ "Timestamp", "Código", "Producto", "Categoría", "Cantidad", "Observaciones" ]); sheet.setFrozenRows(1); sheet.getRange(1, 1, 1, 6).setFontWeight("bold"); } sheet.appendRow([ timestamp, codigo, producto, categoria, cantidad, observaciones ]); return { ok: true }; }
